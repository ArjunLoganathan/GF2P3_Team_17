# ripple_adder.txt
# A 4-bit Ripple Carry Adder implementation using structural primitives.
# Computes: Sum (S3 S2 S1 S0) and Carry-Out (Cout) from Inputs A (A3 A2 A1 A0), B (B3 B2 B1 B0), and Cin.

DEVICES:
    # Input stimulus switches (Initial setup: A = 5 [0101], B = 6 [0110], Cin = 0)
    A0 = SWITCH 1;
    A1 = SWITCH 0;
    A2 = SWITCH 1;
    A3 = SWITCH 0;

    B0 = SWITCH 0;
    B1 = SWITCH 1;
    B2 = SWITCH 1;
    B3 = SWITCH 0;

    Cin = SWITCH 0;

    # --- FULL ADDER 0 (Bit 0) ---
    XA0 = XOR;
    XS0 = XOR;
    AA0 = AND 2;
    AS0 = AND 2;
    CO0 = OR 2;

    # --- FULL ADDER 1 (Bit 1) ---
    XA1 = XOR;
    XS1 = XOR;
    AA1 = AND 2;
    AS1 = AND 2;
    CO1 = OR 2;

    # --- FULL ADDER 2 (Bit 2) ---
    XA2 = XOR;
    XS2 = XOR;
    AA2 = AND 2;
    AS2 = AND 2;
    CO2 = OR 2;

    # --- FULL ADDER 3 (Bit 3) ---
    XA3 = XOR;
    XS3 = XOR;
    AA3 = AND 2;
    AS3 = AND 2;
    CO3 = OR 2;
DEVICES END;

CONNECT:
    # --- Wiring Full Adder 0 ---
    A0  = XA0.I1;
    B0  = XA0.I2;
    XA0 = XS0.I1;
    Cin = XS0.I2;    # Sum 0 is produced at XS0

    A0  = AA0.I1;
    B0  = AA0.I2;
    XA0 = AS0.I1;
    Cin = AS0.I2;
    AA0 = CO0.I1;
    AS0 = CO0.I2;    # Carry 0 (CO0) ripples to FA1

    # --- Wiring Full Adder 1 ---
    A1  = XA1.I1;
    B1  = XA1.I2;
    XA1 = XS1.I1;
    CO0 = XS1.I2;    # Sum 1 is produced at XS1

    A1  = AA1.I1;
    B1  = AA1.I2;
    XA1 = AS1.I1;
    CO0 = AS1.I2;
    AA1 = CO1.I1;
    AS1 = CO1.I2;    # Carry 1 (CO1) ripples to FA2

    # --- Wiring Full Adder 2 ---
    A2  = XA2.I1;
    B2  = XA2.I2;
    XA2 = XS2.I1;
    CO1 = XS2.I2;    # Sum 2 is produced at XS2

    A2  = AA2.I1;
    B2  = AA2.I2;
    XA2 = AS2.I1;
    CO1 = AS2.I2;
    AA2 = CO2.I1;
    AS2 = CO2.I2;    # Carry 2 (CO2) ripples to FA3

    # --- Wiring Full Adder 3 ---
    A3  = XA3.I1;
    B3  = XA3.I2;
    XA3 = XS3.I1;
    CO2 = XS3.I2;    # Sum 3 is produced at XS3

    A3  = AA3.I1;
    B3  = AA3.I2;
    XA3 = AS3.I1;
    CO2 = AS3.I2;
    AA3 = CO3.I1;
    AS3 = CO3.I2;    # Final Carry Out is produced at CO3
CONNECT END;

MONITOR:
    # Monitor the calculated outputs and carry flags
    XS3;   # Sum Bit 3 (MSB)
    XS2;   # Sum Bit 2
    XS1;   # Sum Bit 1
    XS0;   # Sum Bit 0 (LSB)
    CO3;   # Final Carry Out (Cout)
MONITOR END;

END
