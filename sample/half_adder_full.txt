    A;
    B;
INPUT_PORTS END;

    SUM;
    COUT;
OUTPUT_PORTS END;

    B = SWITCH 1;

    A1 = AND 2;

    B = X1.I2;
    A = A1.I1;
    B = A1.I2;
CONNECT END;

    A1;
MONITOR END;
