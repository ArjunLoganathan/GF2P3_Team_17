# comparator_2bit.txt
# Equality comparator for two 2-bit numbers A (A1 A0) and B (B1 B0).
# A bit differs when A_i XOR B_i is high; EQ is high only when no bit differs.

DEVICES:
    A0 = SWITCH 0;
    A1 = SWITCH 0;
    B0 = SWITCH 0;
    B1 = SWITCH 0;

    Diff0 = XOR;        # A0 differs from B0
    Diff1 = XOR;        # A1 differs from B1
    EQ = NOR 2;         # high when both diffs are low
DEVICES END;

CONNECT:
    A0 = Diff0.I1;
    B0 = Diff0.I2;

    A1 = Diff1.I1;
    B1 = Diff1.I2;

    Diff0 = EQ.I1;
    Diff1 = EQ.I2;
CONNECT END;

MONITOR:
    EQ;
    Diff0;
    Diff1;
MONITOR END;

END
