# sample/full_adder.txt
# Reusable Full Adder macro built from two Half Adders.

IMPORT:
    HALFA FROM "sample/half_adder.txt";
IMPORT END;

INPUT_PORTS:
    A;
    B;
    Cin;
INPUT_PORTS END;

OUTPUT_PORTS:
    SUM;
    COUT;
OUTPUT_PORTS END;

DEVICES:
    HA1 = HALFA;
    HA2 = HALFA;
    OR1 = OR 2;
DEVICES END;

CONNECT:
    # First Half Adder adds A and B
    A = HA1.A;
    B = HA1.B;

    # Second Half Adder adds the SUM from HA1 and Cin
    HA1.SUM = HA2.A;
    Cin = HA2.B;

    # Final Carry Out is an OR of the carries from both Half Adders
    HA1.COUT = OR1.I1;
    HA2.COUT = OR1.I2;

    # Route internal signals to the macro's output ports
    HA2.SUM = SUM;
    OR1 = COUT;
CONNECT END;

END
